            def varPPID = transrows.split(',')[2]
            if(it.getPeriodName().equals('BegBalance'))
            {
                def stepnum = 0
                count = 0
                if(!it.crossDimCell('Increment_Date').isMissing())
                {
                    if(!it.crossDimCell('Salary_Plan').isMissing() && !it.crossDimCell('Grade').isMissing())
                    {
                        if(it.crossDimCell('Increment_Date').getDataAsDate() < startDateStr /*&& !varPPID.equals('PP1')*/)
                        {

                            if(!it.crossDimCell('ST_5001',varPPID).isMissing())
                            {
                                /*println "I am inside increment date logic"*/
                                stepnum = it.crossDimCell('Step').getDataAsDouble()
                                count = count+1
                                int stepnumint = (int) stepnum
                                DatatoPrint_SP.each{ sprows ->
                                    if(it.crossDimCell('Salary_Plan').getDataAsSmartListMemberName().equals(sprows.split(',')[12]) && it.crossDimCell('Grade').getDataAsSmartListMemberName().equals(sprows.split(',')[13]))
                                    {
                                        biweeklysal = sprows.split(',')[stepnumint+1].toDouble()
                                        salmax = sprows.split(',')[0].toDouble()
                                    }
                                }
                                if(biweeklysal <= salmax) /*Check if the salary is less than salary max*/
                                {
                                    it.crossDimCell('ST_5001',varPPID).Data = ((it.crossDimCell('Working_Days',varPPID).Data/10) * biweeklysal.toDouble()) * it.crossDimCell('FTE').getData()
                                    it.crossDimCell('IncreSalValue',varPPID).Data = biweeklysal.toDouble()
                                    it.crossDimCell('Step',varPPID).Data = stepnumint+1
                                }
                            }
                        }
                    }
                }
            }
        }
    }
    cube.saveGrid(grid3)
}
