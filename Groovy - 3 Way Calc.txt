def calcTotalMD(List<String> calcMbrs) {
	List<String> dim1 = [] 
	List<String> dim2 = []
	List<String> dim3 = []
	List<String> dim4 = []
	
	calcMbrs.each { Mbrs ->
    	dim1.add(Mbrs.split('-')[0]) ; dim2.add(Mbrs.split('-')[1]) ; dim3.add(Mbrs.split('-')[2]); dim4.add(Mbrs.split('-')[3]);
    }
                  
    List<String> firstMemComb = [dim1[0],dim2[0],dim3[0],dim4[0]]
    List<String> secondMemComb =[dim1[1],dim2[1],dim3[1],dim4[1]]
    List<String> thirdMemComb = [dim1[2],dim2[2],dim3[2],dim4[2]]
    List<String> fourthMemComb = [dim1[3],dim2[3],dim3[3],dim4[3]]
    List<String> fifthMemComb = [dim1[4],dim2[4],dim3[4],dim4[4]]
    
	List<String> EditedDays=[]
    operation.grid.dataCellIterator({DataCell cell -> cell.edited && (cell.getMemberNames().intersect(firstMemComb).size() == 4  || cell.getMemberNames().intersect(secondMemComb).size() == 4 || cell.getMemberNames().intersect(thirdMemComb).size() == 4 || cell.getMemberNames().intersect(fourthMemComb).size() == 4 || cell.getMemberNames().intersect(fifthMemComb).size() == 4 )}).each { cell -> 
   		EditedDays.add(cell.getMemberName("Days"))
    }
    
	operation.grid.dataCellIterator({DataCell cell -> cell.getMemberNames().intersect(firstMemComb).size() == 4  || cell.getMemberNames().intersect(secondMemComb).size() == 4 || cell.getMemberNames().intersect(thirdMemComb).size() == 4 || cell.getMemberNames().intersect(fourthMemComb).size() == 4 || cell.getMemberNames().intersect(fifthMemComb).size() == 4 }).each { cell ->    
        if (cell.getMemberNames().intersect(thirdMemComb).size() == 4 && cell.getMemberName("Days") in EditedDays) {		        	
    	    DataCell firstMemCell = cell.crossDimCell(dim1[0],dim2[0],dim3[0],dim4[0])
        	DataCell secondMemCell = cell.crossDimCell(dim1[1],dim2[1],dim3[1],dim4[1]) 
            DataCell fourthMemCell = cell.crossDimCell(dim1[3],dim2[3],dim3[3],dim4[3])
            DataCell fifthMemCell = cell.crossDimCell(dim1[4],dim2[4],dim3[4],dim4[4])
            	double firstMemData = firstMemCell.data
	    	    double secondMemData = secondMemCell.data
                double fourthMemData = fourthMemCell.data
                double fifthMemData = fifthMemCell.data
                	cell.setForceEditable(true)
			        cell.data = firstMemData + secondMemData + fourthMemData + fifthMemData
                    
        
        }		
	}
    println "${dim1[2]} Calculation Completed"
    
}

//@Method - forumulaCalc - calculate the data based on the formula type provided in 3Way Calc -> R.H.S.Max = 2 members
//Returns - calculated double data type
double formulaCalc (String formulaType,double x,double y) {
	double formCalc
	switch (formulaType) {
    	case "y/x":
	    	formCalc = y / x
	    break;    
    	
        case "x/y":
	    	formCalc = x / y
	    break;  

		case "y*x":
    		formCalc = y * x
	    break;        
    
	    case "y*(1-x)":
    		formCalc = y * (1 - x)
	    break;
        
    	case "y/(1-x)":
    		formCalc = y / (1 - x)
	    break;
       	
        case "1-(y/x)":
			formCalc = 1 - (y/x)             
        break;                
        
        case "(1+x)*y":
        	formCalc = (1+x)*y 
        break;

		case "(y/x)-1":
        	formCalc = (y/x)-1 
        break;

   		default :
    		formCalc = 0
		break
    }
    
    if (formCalc.isNaN() || formCalc.isInfinite()) {
    	formCalc = 0    	
    }
    return formCalc    
}


//@Method - forumulaCalc - calculate the data based on the formula type provided in 3Way Calc -> R.H.S.Max = 3 members
//Returns - calculated double data type
double formulaCalc4Way (String formulaType,double x,double y, double z) {
	double formCalc
	switch (formulaType) {
    	case "y/(x+z)":
	    	formCalc = y / (x + z)
	    break;    
		               
		case "x/(y+z)":
    		formCalc = x / (y + z)
	    break;            	    
        
		case "x*(y+z)":
    		formCalc = x * (y + z)
	    break;          
        
   		default :
    		formCalc = 0
		break
    } 
    if (formCalc.isNaN() || formCalc.isInfinite()) {
    	formCalc = 0    	
    }
    return formCalc    
}

                    cell.setForceEditable(true)
	        		cell.data = formulaCalc(formulaType, firstMemData, secondMemData)

   	       			double tempAllSum = 0

    				periodDesc.each { childPeriod ->

   	            		if (!(cell.crossDimCell(dim1[0],dim2[0],dim3[0],dim4[0],childPeriod.toString()).isLocked())) {

           	            		cell.crossDimCell(childPeriod.toString()).setForceEditable(true)

								}

                	}                    
                }      
			}	 		                
                        
		}
	}
        
    
    operation.grid.dataCellIterator({DataCell cell -> cell.getMemberNames().intersect(thirdMemComb).size() == 4 && (cell.getPeriodName() in parentPeriods) }).each { cell ->

            cell.data = periodSum             
		}		
	}
    println "${dim1[2]} Calculation Completed"        
}
