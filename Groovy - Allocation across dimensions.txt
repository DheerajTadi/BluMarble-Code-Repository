def fiscalTo454Day = { days ->
    String convertedDay ="454_Day0"
    DataGridDefinitionBuilder builder = cube.dataGridDefinitionBuilder()
    builder.addPov(['Years', 'Scenarios', 'Seasons', 'Periods', 'Versions', 'Products', 'Channel'], [['&ActYr'], ['No Scenario'], ['No Season'], ['&CurrMonth'], ['No Version'], ['No Product'], ['No Channel']])
    builder.addColumn(['Metrics'], [ ['WeekDay'] ])
    //builder.addRow(['Days'], [ [days] ])  //commented for groovy upgrade - Vivek
    //Start :: Code change for groovy upgrade - Vivek
    builder.addRow(['Days'], [ [days] ] as List<List<String>>) 
    //End :: Code change for groovy upgrade - Vivek    
    DataGridDefinition gridDefinition = builder.build()
    cube.loadGrid(gridDefinition, false).withCloseable { grid ->
		grid.dataCellIterator().each {
			int a = (int) it.data
			convertedDay += a
		}
	}
	return convertedDay
}
